function loadCategory(category){


const section = data[category];


let html = `<h2>${section.title}</h2>

<p><strong>Is there evidence for these options?</strong> Click each intervention below.</p>`;


section.items.forEach((item,index)=>{


html += `

<div class="card">

<div class="card-header" onclick="toggleCard('${category}-${index}')">

${item.title}

<span class="badge ${item.evidence}">

${item.evidence.toUpperCase()}

</span>

</div>


<div id="${category}-${index}" class="card-content">

${item.text}

</div>

</div>`;

});


document.getElementById("content").innerHTML = html;

}


function toggleCard(id){

const card = document.getElementById(id);


if(card.style.display === "block"){

card.style.display = "none";

}

else{

card.style.display = "block";

}

}
